## 目标

```
       ↓

```

## 智能召回算法

### 召回流程

```javascript
function recall(query, topK = 10) {
  // 1. 解析查询
  const { keywords, techStack, scenario } = parseQuery(query);

  // 2. 分层召回
  let candidates = new Set();

  // Layer 1: 精确匹配
  const exactMatches = searchExact(keywords);
  candidates.addAll(exactMatches);

  // Layer 2: 语义匹配
  const semanticMatches = searchSemantic(keywords);
  candidates.addAll(semanticMatches);

  // Layer 3: 扩展匹配
  const expandedMatches = searchExpanded(keywords);
  candidates.addAll(expandedMatches);

  // Layer 4: 跨域匹配
  const crossMatches = searchCrossDomain(keywords);
  candidates.addAll(crossMatches);

  // Layer 5: 场景推荐
  if (scenario) {
    const scenarioMatches = searchByScenario(scenario);
    candidates.addAll(scenarioMatches);
  }

  // 3. 排序
  const scored = candidates.map(doc => ({
    doc,
    score: calculateScore(doc, query)
  }));

  // 4. 返回 Top K
  return scored.sort((a, b) => b.score - a.score).slice(0, topK);
}
```

### 评分机制

```javascript
function calculateScore(doc, query) {
  let score = 0;

  // 精确匹配 (40%)
  if (doc.title.match(query)) score += 40;
  if (doc.filename.match(query)) score += 30;

  // 关键词匹配 (30%)
  if (doc.keywords.some(k => k.match(query))) score += 20;
  if (doc.synonyms.some(s => s.match(query))) score += 10;

  // 标签匹配 (20%)
  if (doc.tags.some(t => t.match(query))) score += 15;
  if (doc.techStack.some(s => s.match(query))) score += 5;

  // 相关性推荐 (10%)
  if (doc.relatedDocs.some(r => r.match(query))) score += 10;

  return score;
}
```

## 查询优化

### 查询扩展

```javascript
function expandQuery(query) {
  const expansions = [];

  // 1. 添加同义词
  synonyms[query]?.forEach(syn => expansions.push(syn));

  // 2. 添加语义变体
  semanticVariants[query]?.forEach(v => expansions.push(v));

  // 3. 添加技术栈前缀
  if (hasTechStack(query)) {
    expansions.push(`${techStack} ${query}`);
  }

  // 4. 添加场景后缀
  if (hasScenario(query)) {
    expansions.push(`${query} ${scenario}`);
  }

  return [query, ...expansions];
}
```

### 查询改写

```javascript
function rewriteQuery(query) {
  // "动画怎么做" → "动画实现" 或 "动画效果"
  if (query.includes("怎么做")) {
    query = query.replace("怎么做", "实现");
  }

  // "换页" → "页面过渡"
  if (query === "换页") {
    return "页面过渡";
  }

  // "列表很卡" → "列表性能优化"
  if (query.includes("很卡")) {
    return query.replace("很卡", "性能优化");
  }

  return query;
}
```

## 性能优化

### 缓存策略

```javascript
const cache = new Map();

// 缓存热门查询
function cachedSearch(query) {
  if (cache.has(query)) {
    return cache.get(query);
  }

  const result = search(query);
  cache.set(query, result);

  // 限制缓存大小
  if (cache.size > 1000) {
    const firstKey = cache.keys().next().value;
    cache.delete(firstKey);
  }

  return result;
}
```
