# 地图标注渲染引擎

## 概述

本知识介绍一个高性能的地图标注渲染引擎，核心特性：

- **Proxy 响应式**：监听数组变化自动触发渲染
- **throttle 节流**：避免频繁渲染导致的性能问题
- **文字避让**：智能处理标注重叠
- **多类型标注**：支持距离标注、面积标注
- **动态更新**：实时响应地块绘制变化

### 2. 节流渲染

```javascript
// 使用 throttle 避免频繁渲染
const renderAreaAndDistance = throttle(_renderAreaAndDistance, 100);

// throttle 实现
function throttle(func, delay) {
  let lastExecTime = 0;
  let timeoutId = null;
  let lastArgs = null;
  let lastThis = null;

  return function (...args) {
    const currentTime = Date.now();
    const elapsedTime = currentTime - lastExecTime;
    lastArgs = args;
    lastThis = this;

    if (elapsedTime >= delay) {
      if (timeoutId) {
        clearTimeout(timeoutId);
        timeoutId = null;
      }
      func.apply(this, args);
      lastExecTime = currentTime;
    } else if (!timeoutId) {
      timeoutId = setTimeout(() => {
        func.apply(lastThis, lastArgs);
        lastExecTime = Date.now();
        timeoutId = null;
      }, delay - elapsedTime);
    }
  };
}
```

### 3. 主渲染函数

```javascript
function _renderAreaAndDistance(points, map) {
  drawLineLabelLayer.clearLayers();

  // 缩放级别过小时不显示标注
  if (map.getZoom() < 12) return;

  if (points.length < 1) return;

  // 渲染距离标注
  const markerLayers = getDistanceLabelMarker(JSON.parse(JSON.stringify(points)), map);
  if (!(markerLayers && markerLayers.length)) return;

  markerLayers.map(layer => {
    drawLineLabelLayer.addLayer(layer);
  });

  // 处理标注重叠
  const spanList = document.querySelectorAll('.line-label-icon span.line-label-icon__content');
  let hideNode = optimizeSpanRendering(spanList);

  // 重叠过多时隐藏全部
  if (hideNode / spanList.length > 0.85) {
    drawLineLabelLayer.clearLayers();
  }

  // 闭合后渲染面积标注
  const isClose = points.length > 3 &&
    (points[0][0].toString() === points[points.length - 1][0].toString() &&
     points[0][1].toString() === points[points.length - 1][1].toString());

  if (isClose) {
    const areaLayer = getAreaLabelMarker(points);
    if (!areaLayer) return;
    drawLineLabelLayer.addLayer(areaLayer);
  }
}
```

## 距离标注

### 1. 获取距离标注

```javascript
function getDistanceLabelMarker(points, map) {
  if (points.length < 1) return;

  let textLabelMarkerLayers = [];

  for (let i = 0; i < points.length - 1; i++) {
    const latlng1 = points[i];
    const latlng2 = points[i + 1];

    // 计算中点坐标
    const midpoint = turf.midpoint(
      turf.point([latlng1[1], latlng1[0]]),
      turf.point([latlng2[1], latlng2[0]])
    );

    const midLatLng = {
      lat: midpoint.geometry.coordinates[1],
      lng: midpoint.geometry.coordinates[0],
    };

    // 计算角度
    const angle = calculateAngle({ lat: latlng1[0], lng: latlng1[1] }, midLatLng);

    // 计算标注偏移位置（避免在图形内部）
    const [offsetLatLng1, offsetLatLng2] = calculateDestination(midLatLng, 15, angle);

    // 计算距离
    let distance = map.distance(latlng1, latlng2);
    distance = numTofixed(distance, 1);

    // 判断坐标是否在图形内部
    const isInside = isPointInPolygon(offsetLatLng1, points);

    // 计算文字旋转角度
    let textRotate = angle;
    if (angle > 90 && angle < 270) {
      textRotate += 180;
    }

    // 创建标注
    textLabelMarkerLayers.push(
      L.marker(isInside ? offsetLatLng2 : offsetLatLng1, {
        icon: L.divIcon({
          iconSize: [20, 20],
          className: 'line-label-icon',
          html: `<span class="line-label-icon__content"
            style="transform: translate(0, 0) rotate(${textRotate}deg);">
            ${distance === 0 ? '' : `${numTofixed(distance, 1)}米`}
          </span>`,
        }),
      })
    );
  }

  return textLabelMarkerLayers;
}
```

### 2. 计算两点间角度

```javascript
function calculateAngle(latlng1, latlng2) {
  const R = 6371000; // 地球半径（米）
  const dLng = ((latlng2.lng - latlng1.lng) * Math.PI) / 180;
  const dLat = ((latlng2.lat - latlng1.lat) * Math.PI) / 180;

  // 笛卡尔坐标转换
  const dx = R * dLng * Math.cos(((latlng1.lat + latlng2.lat) * Math.PI) / 360);
  const dy = R * dLat;

  // 计算相对于正东方向的夹角
  let angleRad = Math.atan2(dy, dx);
  let angleDeg = (angleRad * -180) / Math.PI;

  if (angleDeg < 0) {
    angleDeg += 360;
  }

  return angleDeg;
}
```

### 3. 坐标偏移计算

```javascript
function calculateDestination(latlng, distance, bearing) {
  // 经纬度转换为屏幕像素
  const point = geomanDraw.map.latLngToContainerPoint(latlng);
  const absolutePoint = { x: 0, y: 0 };

  let transBearing = bearing;

  // 根据象限计算偏移
  if (bearing > 0 && bearing <= 90) {
    // 第一象限
    absolutePoint.x = distance * Math.sin(toRadians(transBearing));
    absolutePoint.y = bearing === 90 ? 0 : distance * Math.cos(toRadians(transBearing)) * -1;
  } else if (bearing > 90 && bearing <= 180) {
    // 第二象限
    transBearing -= 90;
    absolutePoint.x = distance * Math.cos(toRadians(transBearing));
    absolutePoint.y = bearing === 180 ? 0 : distance * Math.sin(toRadians(transBearing));
  } else if (bearing > 180 && bearing <= 270) {
    // 第三象限
    transBearing -= 180;
    absolutePoint.x = distance * Math.sin(toRadians(transBearing)) * -1;
    absolutePoint.y = bearing === 270 ? 0 : distance * Math.cos(toRadians(transBearing));
  } else {
    // 第四象限
    transBearing -= 270;
    absolutePoint.x = distance * Math.cos(toRadians(transBearing)) * -1;
    absolutePoint.y = bearing === 0 ? 0 : distance * Math.sin(toRadians(transBearing)) * -1;
  }

  // 返回两个镜像坐标点
  return [
    geomanDraw.map.containerPointToLatLng({
      x: point.x + absolutePoint.x,
      y: point.y + absolutePoint.y,
    }),
    geomanDraw.map.containerPointToLatLng({
      x: point.x - absolutePoint.x,
      y: point.y - absolutePoint.y,
    }),
  ];
}

function toRadians(degrees) {
  return degrees * (Math.PI / 180);
}
```

## 面积标注

### 1. 获取面积标注

```javascript
function getAreaLabelMarker(points, options = {}) {
  if (!points) return;

  let relatedPinter;

  // 处理不同格式的坐标
  if (points.length === 2 && !isNaN(points[0]) && !isNaN(points[1])) {
    // 二维坐标
    relatedPinter = points;
  } else {
    // 多边形坐标
    if (points.length < 3) return;
    relatedPinter = getCenterInPolygon(JSON.parse(JSON.stringify(points)));
  }

  // 计算面积字符串
  let areaStr = 0;
  if (options.areaMu) {
    areaStr = `${options.areaMu}亩`;
  } else {
    let { area, unitName } = calculatePolygonArea(points);
    areaStr = `${area}${unitName}`;
  }

  return L.marker(relatedPinter, {
    icon: L.divIcon({
      iconSize: [20, 20],
      className: 'land-area-label',
      html: `${options.landName ? `<span class="land-area-label-content">${options.landName}</span>` : ``}<span class="land-area-label-content">${areaStr}</span>`,
    }),
    path: points,
    ...options,
  });
}
```

### 2. 计算多边形中心点

```javascript
function getCenterInPolygon(rings) {
  const polygon = {
    type: 'Polygon',
    coordinates: [rings],
  };

  // 1. 尝试质心
  const massCenter = turf.centerOfMass(polygon);
  if (turf.booleanPointInPolygon(turf.point(massCenter.geometry.coordinates), polygon)) {
    return massCenter.geometry.coordinates;
  }

  // 2. 尝试几何中心
  const center = turf.centroid(polygon);
  if (turf.booleanPointInPolygon(turf.point(center.geometry.coordinates), polygon)) {
    return center.geometry.coordinates;
  }

  // 3. 尝试最大内切圆中心
  var maxInscribedCircleCenter = relatedpointer.indexfn(rings, 21);
  if (turf.booleanPointInPolygon(turf.point(maxInscribedCircleCenter), polygon)) {
    return maxInscribedCircleCenter;
  }

  // 4. 保底：取第一个点
  return Array.isArray(rings[0][0][0]) ? rings[0][0][0] :
         Array.isArray(rings[0][0]) ? rings[0][0] : rings[0];
}
```

### 3. 计算多边形面积

```javascript
function calculatePolygonArea(polygonPoints) {
  if (!Array.isArray(polygonPoints) || polygonPoints.length < 3) {
    return 0;
  }

  // 确保多边形闭合
  const firstPoint = polygonPoints[0];
  const lastPoint = polygonPoints[polygonPoints.length - 1];
  if (firstPoint[0] !== lastPoint[0] || firstPoint[1] !== lastPoint[1]) {
    polygonPoints = [...polygonPoints, firstPoint];
  }

  try {
    // 转换为 [经度, 纬度] 格式
    const turfPolygon = turf.polygon([
      polygonPoints.map(point => [point[1], point[0]])
    ]);
    const { area, unitName } = getAreaStr(turfPolygon);
    return { area: parseFloat(area.toFixed(1)), unitName };
  } catch (error) {
    console.error('计算面积出错:', error);
    return 0;
  }
}

// 获取带单位的面积字符串
function getAreaStr(polygon, keepDecimal = 1) {
  let area = 0;
  let areaUnit = 3;
  var unitarr = ['', '亩', '分', '平方米'];

  const squareMeterArea = turf.area(polygon);

  // 自动选择单位
  if (squareMeterArea > 666.7) {        // > 1亩
    area = numTofixed(squareMeterAreaTranlate(squareMeterArea, 1), keepDecimal);
    areaUnit = 1;
  } else if (squareMeterArea > 66.67) { // > 1分
    area = numTofixed(squareMeterArea / 66.67, keepDecimal);
    areaUnit = 2;
  } else {
    area = numTofixed(squareMeterArea, keepDecimal);
    areaUnit = 3;
  }

  return { area, areaUnit, unitName: unitarr[areaUnit] };
}
```

## 文字避让算法

### 1. 分离轴定理（SAT）

```javascript
function optimizeSpanRendering(nodeList) {
  const keptNodes = [];

  // 计算旋转后的矩形顶点
  function getRotatedCorners(el) {
    const rect = el.getBoundingClientRect();
    const centerX = rect.left + rect.width / 2;
    const centerY = rect.top + rect.height / 2;

    const style = window.getComputedStyle(el);
    let angle = 15.61;
    if (style.transform && style.transform !== 'none') {
      const matrix = new DOMMatrix(style.transform);
      angle = Math.atan2(matrix.b, matrix.a) * (180 / Math.PI);
    }

    const rad = (angle * Math.PI) / 180;
    const cos = Math.cos(rad);
    const sin = Math.sin(rad);

    // 返回旋转后的四个角点
    return [
      { x: centerX + (rect.width / 2) * cos - (rect.height / 2) * sin,
        y: centerY + (rect.width / 2) * sin + (rect.height / 2) * cos },
      { x: centerX - (rect.width / 2) * cos - (rect.height / 2) * sin,
        y: centerY - (rect.width / 2) * sin - (rect.height / 2) * cos },
      { x: centerX - (rect.width / 2) * cos + (rect.height / 2) * sin,
        y: centerY - (rect.width / 2) * sin + (rect.height / 2) * cos },
      { x: centerX + (rect.width / 2) * cos + (rect.height / 2) * sin,
        y: centerY + (rect.width / 2) * sin - (rect.height / 2) * cos },
    ];
  }

  // 检测两个多边形是否重叠
  function checkPolygonOverlap(polyA, polyB) {
    const polygons = [polyA, polyB];
    for (let i = 0; i < polygons.length; i++) {
      const polygon = polygons[i];
      for (let j = 0; j < polygon.length; j++) {
        const p1 = polygon[j];
        const p2 = polygon[(j + 1) % polygon.length];
        const normal = { x: p2.y - p1.y, y: p1.x - p2.x };

        const projectPolygon = (poly) => {
          let min = Infinity;
          let max = -Infinity;
          for (const point of poly) {
            const projected = point.x * normal.x + point.y * normal.y;
            min = Math.min(min, projected);
            max = Math.max(max, projected);
          }
          return { min, max };
        };

        const proj1 = projectPolygon(polyA);
        const proj2 = projectPolygon(polyB);

        // 存在分离轴，没有碰撞
        if (proj1.max < proj2.min || proj2.max < proj1.min) {
          return false;
        }
      }
    }
    return true; // 所有轴都重叠，发生碰撞
  }

  // 处理每个节点
  let hideNodeList = 0;
  Array.from(nodeList).forEach(currentNode => {
    const currentCorners = getRotatedCorners(currentNode);
    let canKeep = true;

    for (const keptNode of keptNodes) {
      const keptCorners = getRotatedCorners(keptNode);
      if (checkPolygonOverlap(currentCorners, keptCorners)) {
        canKeep = false;
        break;
      }
    }

    if (canKeep) {
      keptNodes.push(currentNode);
    } else {
      hideNodeList++;
      currentNode.textContent = ''; // 清空重叠节点
    }
  });

  return hideNodeList;
}
```
