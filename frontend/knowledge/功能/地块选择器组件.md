# 地块选择器组件

## 概述

本知识介绍一个完整的 Vue 2.x 地块选择器组件实现方案，支持：

- 地块多选
- 地块分组管理
- 地图选地块弹框
- 面积自动统计
- 搜索过滤

## 组件结构

### 1. Vue 组件注册

```javascript
var landSelectorTemp = `
  <div class='land-selector-comp special' ref="landSelect">
    <span class='text addbtn'>
      <i v-show='selectList.length===0' class='tips addbtn'>请选择地块</i>
      <i v-show='selectList.length>0' class='selected addbtn'>
        已选择 <i class='num'>{{selectList.length}}</i> 个地块
      </i>
    </span>
    <span class="edge-icon-animate" :class="{active: isDrop}">
      <slot name="icon"><i class='edge-icon'></i></slot>
    </span>
  </div>
`;

Vue.component('land-selector', {
  template: landSelectorTemp,
  props: {
    showMap: { type: Boolean, default: false },
    hideCropList: { type: Boolean, default: true },
    selectedLandList: { type: Array, default: [] },
  },
  data() {
    return {
      selectList: [],
      tempSelectList: [],
      isDrop: false,
      areaSelectEl: null,
    };
  },
  async mounted() {
    this.selectList = this.tempSelectList = JSON.parse(JSON.stringify(this.selectedLandList));
    await this.$nextTick();
    this.init();
  },
  methods: {
    init() {
      this.initPlotSelectfn(
        this.selectedLandList,
        (selectList, totalArea, areaList, doSearch = true) => {
          // 更新选择状态
          this.selectList = selectList;
          // 更新显示文本
          var selectAreaHtml = selectList.length > 0
            ? `已选择 <i class="num">${selectList.length}</i>个地块`
            : `<i class="tips addbtn">请选择地块</i>`;
          $('.land-selector-comp span.addbtn').html(selectAreaHtml);

          // 选中未改变不向上触发事件
          if (JSON.stringify(this.tempSelectList.sort()) === JSON.stringify(this.selectList.sort())) return;

          if (doSearch) {
            this.$emit('change', selectList);
            this.tempSelectList = selectList;
          }
        },
        this.showMap,
        this.hideCropList
      );
    },
  },
});
```

initPlotSelectfn(selectIds, nextfn, showMap = false, hideCropList = true) {
  var landList = [];

    showAllSelect: true,
    getListfn: getPlotArea,

    // 单次选择回调
    callback: function (landlist) {
      reflashResult(landlist, false);
    },

    // 弹框关闭回调
    closeCallback: function (landlist) {
      landList = landlist;
      reflashResult(landList, true);
      that.isDrop = false;
    },

    // 弹框打开回调
    openCallback: function () {
      that.isDrop = true;
      that.$emit('openCallback');
    },
  });
}

### 3. 地块数据处理

```javascript
async function getPlotArea(nextfn) {
  // 获取地块列表
  var [groupListResult, groupListPathResult] = await getLandList();
  var landIdMapPath = {};

  // 构建 ID 到路径的映射
  groupListPathResult.map(item => {
    landIdMapPath[item['id']] = item;
  });

  // 合并地块数据
  groupListResult.forEach(groupItem => {
    groupItem['children'] && groupItem['children'].forEach(landItem => {
      var landid = landItem['id'];
      landItem = Object.assign(landItem, landIdMapPath[landid]);
    });
  });

  // 构建选择器数据格式
  var arealist = [];
  groupListResult.forEach(function (groupItem, index) {
    if (groupItem.children.length === 0) return;

    var groupName = groupItem.name;
    var groupId = groupItem.id;
    var groupOpen = false;

    // 默认编组展开
    if (groupItem.id === 0) {
      groupOpen = true;
      if (groupListResult.length === 1) {
        groupName = '全部地块';
      }
    }

    // 添加分组项
    arealist.push({
      itemId: groupId,
      name: groupName,
      groupId: groupId,
      selected: false,
      isGroup: true,
      isOpen: groupOpen,
      landList: groupItem.children.map(item => item.id),
      data: groupItem.children,
    });

    // 添加地块项
    groupItem.children.forEach(landItem => {
      arealist.push({
        itemId: landItem.id,
        name: landItem.landName,
        groupId: groupId,
        selected: false,
        isGroup: false,
        isShow: groupOpen,
        data: landItem,
      });
    });
  });

  // 回显已选项
  if (selectIds && selectIds.length) {
    arealist.map(area => {
      if (selectIds.includes(area.itemId) && !area.isGroup) {
        area.selected = true;
      }
      // 全选分组
      if (area.isGroup) {
        var hasAllId = true;
        area.landList.map(id => {
          if (!selectIds.includes(id)) {
            hasAllId = false;
          }
        });
        if (hasAllId) {
          area.selected = true;
        }
      }
    });
  }

  landList = arealist;
  nextfn(arealist);
}
```

### 4. 结果处理

```javascript
function reflashResult(selectlist, doSearch = true) {
  landSelectedIds = [];
  acreTotal = 0;

  selectlist.forEach(item => {
    if (!item.isGroup && item.selected) {
      landSelectedIds.push(item.itemId);
      // 面积累加（单位转换）
      let tempAcre = area2Mu(item.data.landAcre, item.data.landAcreUnit, false);
      acreTotal += tempAcre;
    }
  });

  acreTotal = parseFloat(acreTotal.toFixed(2));
  nextfn && nextfn(landSelectedIds, acreTotal, landList, doSearch);
}

// 面积单位转换
function area2Mu(area, areaUnit, isFixedTo2 = true) {
  switch (areaUnit) {
    case 3: // 平方米转亩
      return isFixedTo2
        ? parseFloat((area * 0.0015).toFixed(2))
        : parseFloat(area * 0.0015);
    case 2: // 分转亩
      return isFixedTo2
        ? parseFloat((area * 0.1).toFixed(2))
        : parseFloat(area * 0.1);
    case 1:
    default:
      return parseFloat(area);
  }
}
```

## 使用方式

### 1. 组件使用

```vue
<land-selector
  :selectedLandList="selectedLandIds"
  :showMap="true"
  @change="handleLandChange"
  @openCallback="handleOpen"
/>
```

### 2. Props

| 属性 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| selectedLandList | Array | [] | 已选择的地块ID列表 |
| showMap | Boolean | false | 是否显示地图选地块 |
| hideCropList | Boolean | true | 是否隐藏作物列表 |

### 3. Events

| 事件 | 参数 | 说明 |
|------|------|------|
| change | selectList | 选择改变时触发 |
| openCallback | - | 弹框打开时触发 |

## 常见问题

### 1. 面积单位不一致

```
问题：不同地块面积单位不同

解决：统一转换为亩
area2Mu(area, areaUnit, false)
```

### 2. 选中状态不同步

```
问题：选择地块后状态不同步

原因：比较逻辑问题

解决：使用 JSON.stringify 深度比较
if (JSON.stringify(this.tempSelectList.sort()) === JSON.stringify(this.selectList.sort())) return;
```

### 3. 地图选地块数据传递

```
问题：地图选择的数据无法回传

解决：使用回调函数传递选中ID
saveCallback(selectedLandIds)
```
